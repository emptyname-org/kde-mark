cmake_minimum_required(VERSION 3.20)
project(dolphin_mark VERSION 0.1.1 LANGUAGES CXX)

include(GNUInstallDirs)
include(CTest)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_INSTALL_DEFAULT_DIRECTORY_PERMISSIONS
    OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)

find_package(Qt6 6.4 REQUIRED COMPONENTS Core DBus)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
find_program(MARK_PYTHON NAMES python3 REQUIRED)
# Only KIOCore is needed; avoid pulling in KIOWidgets' development dependencies.
find_path(KIO_INCLUDE_DIR KOverlayIconPlugin PATH_SUFFIXES KF6/KIOCore REQUIRED)
find_path(KIO_VERSION_INCLUDE_DIR kio_version.h PATH_SUFFIXES KF6/KIO KF6 REQUIRED)
find_library(KIO_LIBRARY NAMES KF6KIOCore REQUIRED)
set(MARK_PLUGIN_DIR "${CMAKE_INSTALL_LIBDIR}/qt6/plugins/kf6/overlayicon" CACHE PATH "Qt 6 overlay plugin directory")

execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_icons.py"
    "${CMAKE_CURRENT_BINARY_DIR}/icons" COMMAND_ERROR_IS_FATAL ANY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS tools/generate_icons.py src/mark.py)
add_library(dolphin_mark MODULE src/overlay.cpp src/overlay.h "${CMAKE_CURRENT_BINARY_DIR}/icons/icons.qrc")
target_include_directories(dolphin_mark SYSTEM PRIVATE "${KIO_INCLUDE_DIR}" "${KIO_VERSION_INCLUDE_DIR}")
target_link_libraries(dolphin_mark PRIVATE Qt6::Core Qt6::DBus "${KIO_LIBRARY}")
target_compile_options(dolphin_mark PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_target_properties(dolphin_mark PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/plugins/kf6/overlayicon")

configure_file(assets/mark.desktop.in mark.desktop @ONLY)
configure_file(src/mark.py share/kde-mark/mark.py COPYONLY)
install(TARGETS dolphin_mark LIBRARY DESTINATION "${MARK_PLUGIN_DIR}")
install(PROGRAMS src/mark.py DESTINATION "${CMAKE_INSTALL_DATADIR}/kde-mark")
install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/mark.desktop" DESTINATION "${CMAKE_INSTALL_DATADIR}/kio/servicemenus")
file(GLOB menu_icons "${CMAKE_CURRENT_BINARY_DIR}/icons/mark-color-*.svg")
install(FILES ${menu_icons} DESTINATION "${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/emblems")
install(FILES LICENSE DESTINATION "${CMAKE_INSTALL_DATADIR}/doc/kde-mark" RENAME copyright)

if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(test_overlay tests/test_overlay.cpp)
    target_include_directories(test_overlay SYSTEM PRIVATE "${KIO_INCLUDE_DIR}" "${KIO_VERSION_INCLUDE_DIR}")
    target_link_libraries(test_overlay PRIVATE Qt6::Core Qt6::DBus Qt6::Test "${KIO_LIBRARY}")
    target_compile_options(test_overlay PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_dependencies(test_overlay dolphin_mark)
    add_test(NAME overlay COMMAND test_overlay "$<TARGET_FILE:dolphin_mark>")
    add_test(NAME metadata COMMAND "${MARK_PYTHON}" -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/tests" -p "test_*.py")
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/tmp")
    set_tests_properties(overlay metadata PROPERTIES
        TIMEOUT 40
        ENVIRONMENT "TMPDIR=${CMAKE_CURRENT_BINARY_DIR}/tmp;XDG_DATA_HOME=${CMAKE_CURRENT_BINARY_DIR}/share;QT_QPA_PLATFORM=offscreen;PYTHONDONTWRITEBYTECODE=1")
endif()

set(CPACK_PACKAGE_NAME "kde-mark")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_CONTACT "emptyname <noreply@emptyname.org>")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Mark files and folders with coloured icon badges in Dolphin")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
set(CPACK_DEBIAN_PACKAGE_SECTION "kde")
set(CPACK_DEBIAN_PACKAGE_HOMEPAGE "https://github.com/emptyname-org/kde-mark")
set(CPACK_DEBIAN_PACKAGE_DEPENDS "dolphin, python3 (>= 3.10), python3-pyqt6")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_STRIP_FILES ON)
include(CPack)
